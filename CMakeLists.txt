cmake_minimum_required(VERSION 3.21)
project(SerialStudio VERSION 0.4.2 LANGUAGES CXX)
set(SERIALSTUDIO_VERSION "0.4.2-beta.1")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps/qt/lib/cmake/Qt6SerialPort/Qt6SerialPortConfig.cmake")
    set(Qt6SerialPort_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps/qt/lib/cmake/Qt6SerialPort" CACHE PATH "工程内串口模块")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 SerialPort Network Multimedia Test QuickTest)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(SerialStudio WIN32 src/main.cpp)
set_source_files_properties(qml/components/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(SerialStudio
    URI SerialWorkbench
    VERSION 1.0
    QML_FILES qml/Main.qml
        qml/components/Theme.qml qml/components/ActionButton.qml
        qml/components/SelectBox.qml qml/components/InputBox.qml
        qml/components/Panel.qml qml/components/Metric.qml
        qml/pages/TerminalPage.qml qml/pages/ProtocolPage.qml qml/pages/SettingsPage.qml
        qml/pages/NetworkConfig.qml qml/pages/BoardPage.qml
        qml/pages/DataPanel.qml qml/pages/TransferPage.qml qml/pages/CommandAssistant.qml
)
target_link_libraries(SerialStudio PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::SerialPort Qt6::Multimedia transfercore)
if(MINGW)
    target_compile_options(SerialStudio PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_library(updatecore STATIC src/updatecontroller.cpp src/updatecontroller.h)
target_include_directories(updatecore PUBLIC src)
target_link_libraries(updatecore PUBLIC Qt6::Network Qt6::Gui)
target_link_libraries(SerialStudio PRIVATE updatecore)
target_compile_definitions(SerialStudio PRIVATE SERIALSTUDIO_VERSION="${SERIALSTUDIO_VERSION}")
qt_add_executable(update_tests tests/update_tests.cpp)
target_link_libraries(update_tests PRIVATE updatecore Qt6::Test)
add_test(NAME update_tests COMMAND update_tests)
if(MINGW)
    target_compile_options(updatecore PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(terminalcore STATIC src/terminalcontroller.cpp src/terminalcontroller.h src/adbdiscovery.cpp src/adbdiscovery.h)
target_include_directories(terminalcore PUBLIC src)
target_link_libraries(terminalcore PUBLIC Qt6::Core)
qt_add_resources(terminalcore terminal_catalog PREFIX "/terminal" FILES data/terminal_commands.json)
qt_add_executable(terminal_tests tests/terminal_tests.cpp)
target_link_libraries(terminal_tests PRIVATE terminalcore Qt6::Test)
add_test(NAME terminal_tests COMMAND terminal_tests -o terminal-results.txt,txt)
add_library(datacore STATIC src/payloaddecoder.cpp src/payloaddecoder.h src/dataworkspace.cpp src/dataworkspace.h)
target_include_directories(datacore PUBLIC src)
target_link_libraries(datacore PUBLIC Qt6::Gui)
qt_add_executable(data_tests tests/data_tests.cpp)
target_link_libraries(data_tests PRIVATE datacore Qt6::Test)
add_test(NAME data_tests COMMAND data_tests)
qt_add_executable(media_tests tests/media_tests.cpp)
target_link_libraries(media_tests PRIVATE datacore Qt6::Multimedia Qt6::Test)
add_test(NAME media_tests COMMAND media_tests)
set_tests_properties(media_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_library(transfercore STATIC src/filetransfer.cpp src/filetransfer.h)
target_include_directories(transfercore PUBLIC src)
target_link_libraries(transfercore PUBLIC Qt6::Core)
qt_add_executable(transfer_tests tests/transfer_tests.cpp)
target_link_libraries(transfer_tests PRIVATE transfercore Qt6::Test)
add_test(NAME transfer_tests COMMAND transfer_tests)
add_library(protocol STATIC src/protocol.cpp src/protocol.h src/bridgecodec.cpp src/bridgecodec.h)
target_link_libraries(protocol PUBLIC Qt6::Core)
target_include_directories(protocol PUBLIC src)
qt_add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE protocol Qt6::Test)
add_test(NAME core_tests COMMAND core_tests)
add_library(sessioncore STATIC src/session.cpp src/session.h src/serialworker.cpp src/serialworker.h src/eventmodel.cpp src/eventmodel.h)
target_link_libraries(sessioncore PUBLIC protocol datacore Qt6::SerialPort)
target_link_libraries(SerialStudio PRIVATE sessioncore)
qt_add_executable(session_tests tests/session_tests.cpp)
target_link_libraries(session_tests PRIVATE sessioncore Qt6::Test)
add_test(NAME session_tests COMMAND session_tests)
add_library(networkcore STATIC src/networkworker.cpp src/networkworker.h)
target_link_libraries(networkcore PUBLIC Qt6::Network)
target_include_directories(networkcore PUBLIC src)
target_link_libraries(sessioncore PUBLIC networkcore)
add_library(boardcore STATIC src/boardcontroller.cpp src/boardcontroller.h)
target_include_directories(boardcore PUBLIC src)
target_link_libraries(boardcore PUBLIC sessioncore terminalcore)
qt_add_resources(boardcore board_scripts PREFIX "/board" FILES board/board_agent.py)
target_link_libraries(SerialStudio PRIVATE boardcore)
qt_add_executable(board_tests tests/board_tests.cpp)
target_link_libraries(board_tests PRIVATE boardcore Qt6::Test)
add_test(NAME board_tests COMMAND board_tests)
qt_add_executable(network_tests tests/network_tests.cpp)
target_link_libraries(network_tests PRIVATE networkcore Qt6::Test)
add_test(NAME network_tests COMMAND network_tests)
qt_add_executable(ui_tests tests/ui_main.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/.qt/rcc/qrc_qmake_SerialWorkbench.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/.qt/rcc/qrc_SerialStudio_raw_qml_0.cpp
    ${CMAKE_CURRENT_BINARY_DIR}/.qt/rcc/qrc_SerialStudio_raw_qml_0_extra_qmldirs.cpp)
add_dependencies(ui_tests SerialStudio)
target_link_libraries(ui_tests PRIVATE Qt6::QuickTest Qt6::QuickControls2)
add_test(NAME ui_tests COMMAND ui_tests -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/qml)
set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_FONTDIR=C:/Windows/Fonts")
if(MINGW)
    target_compile_options(terminalcore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(datacore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(transfercore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(protocol PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(sessioncore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(networkcore PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(boardcore PRIVATE -Wall -Wextra -Wpedantic)
endif()
